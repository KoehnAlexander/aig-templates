\begin{frame}{What is Possibilistic Logic?}
	\begin{block}{Possibilistic logic}
		Possibilistic logic is a weighted logic that handles uncertainty in a qualitative way by adding certainty to classical logic formulas.
	\end{block}
	\begin{itemize}
		\item \textbf{Applications}: Useable in expert systems, decision-making, and AI for reasoning under uncertainty.
	\end{itemize}

\section{Possibility Theory}
% Now I will introduce you first into Possibilty theory
% Slide 2:  
\begin{frame}{Possibility Distribution}
	\begin{block}{Possibility distribution}
		A \textbf{possibility distribution} is a mapping:
		\[
		\pi: U \to S,
		\]
		where:
		\begin{itemize}
			\item \( U \): Set of states
			\item \( S \): Totally ordered scale (\( S = [0, 1] \) or finite chains).
			\item \( \pi(u) \): Degree of possibility (plausibility) of state \( u \).
		\end{itemize}
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item \( \pi(u) = 0 \): State \( u \) is \textbf{impossible}.
			\item \( \pi(u) = 1 \): State \( u \) is \textbf{fully possible}.
		\end{itemize}
	\end{exampleblock}
	\begin{block}{Normalization Condition}
		At least one state \( u \in U \) must have \( \pi(u) = 1 \) (an actual state exists).
	\end{block}
\end{frame}

% To start with possibility theory we need to first understand possibility distributions
% A possibility distribution is a mapping ? from U to a totally ordered scale S, with top denoted by 1 and bottom by 0

\begin{frame}{Specificity of a Possibility Distribution}
	\begin{block}{Specificity of a Possibility Distribution}
		A distribution \( \pi \) is \textbf{at least as specific} as \( \pi' \) if:
		\[
		\pi(u) \leq \pi'(u), \quad \forall u \in U.
		\]
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item \textbf{Complete knowledge:} \( \pi(u_0) = 1 \), \( \pi(u) = 0 \) for all \( u \neq u_0 \).
			\item \textbf{Complete ignorance:} \( \pi(u) = 1 \) for all \( u \in U \).
		\end{itemize}
	\end{exampleblock}
	\begin{block}{Principle of minimal specificity}
		It states that any hypothesis not known to be impossible cannot be ruled out. It is a minimal commitment, cautious, information principle.
	\end{block}
\end{frame}
% Notes:
% A possibility distribution ? is said to be at least as specific as another ??if and only if for each state of a?airs u: ?(u) ???(u) Then, ? is at least as restrictive and informative as ??, since it rules out at least as many states with at least as much strength.

% Slide 4:
\begin{frame}{Possibility and Necessity Functions}
	\begin{block}{Possibility measure}
		\[
		\Pi(A) = \sup_{u \in A} \pi(u).
		\]
		Evaluates the extent to which \( A \) is consistent with \( \pi \).
	\end{block}
	\begin{block}{Necessity measure}
		\[
		N(A) = \inf_{u \notin A} (1 - \pi(u)).
		\]
		Evaluates the extent to which \( A \) is certainly implied by \( \pi \).
	\end{block}
	
\end{frame}
% Notes:
% Given a simple query of the form ?does event A occur?? 
% is the corresponding proposition a true?
% A is a subset of states, the set of models of a, 
% the response to the query can be obtained by computing degrees of possibility and necessity respectively

% Slide 5:
\begin{frame}{Key Properties of Measures}
	\begin{block}{Duality Relationship}
		\[
		N(A) = 1 - \Pi(A^c).
		\]
		Where \( A^c \) is the complement of \( A \).
	\end{block}
	\begin{block}{Possibility: Maxitivity Property}
		\[
		\Pi(A \cup B) = \max(\Pi(A), \Pi(B)).
		\]
	\end{block}
	\begin{block}{Necessity: Minimization Property}
		\[
		N(A \cap B) = \min(N(A), N(B)).
		\]
	\end{block}
\end{frame}
% Notes:
% Possibility measures satisfy the characteristic ?maxitivity? property
% Necessity measures satisfy an axiom dual to that of possibility measures
% N(a?b) = min(N(a),N(b)), expresses that to be certain about a?b, one has to be certain about a and to be certain about b.
% Note also that we only have N(A?B) ? max(N(A),N(B)).This goes well with the idea that one may be certain about the event A?B, without being really certain about more specific events such as A and B.

% Slide 6:
\begin{frame}{Certainty Qualification in Possibilistic Logic}
	\begin{itemize}
		\item Certainty-qualified information: "A is certain to degree \(\alpha\)" is modeled by the constraint: 
		\[
		N(A) \geq \alpha
		\]
		\item Least Specific possibility distribution:
		\[
		\pi_{(A, \alpha)}(u) = 
		\begin{cases} 
			1 & \text{if } u \in A, \\ 
			1 - \alpha & \text{otherwise}.
		\end{cases}
		\]
	\end{itemize}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item If \(\alpha = 1\): \(\pi(A, \alpha)\) becomes the characteristic function of \(A\).
			\item If \(\alpha = 0\): \(\pi(A, \alpha)\) represents total ignorance.
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Notes:
% Human knowledge is often expressed as statements with degrees of belief.
% Represents a family of possible epistemic states \(\pi\) that satisfy this constraint.
% Integral to the semantics of possibilistic logic.

% Slide 7:
\begin{frame}{Example}
	\begin{itemize}
		\item A doctor is trying to diagnose a patient based on symptoms and diseases:
		\begin{itemize}
			\item \( U = \{\text{flu}, \text{cold}, \text{allergy}\} \).
			\item Possibility distribution \(\pi\) based on symptom severity:
			\[
			\pi(\text{flu}) = 0.9, \quad \pi(\text{cold}) = 0.6, \quad \pi(\text{allergy}) = 0.3.
			\]
		\end{itemize}
		\item The doctor evaluates the possibility and necessity measures for a hypothesis \( A \):
		\[ 
		A = \{\text{flu}, \text{cold}\}
		\]
		\item Calculation:
		\begin{align*}
			\Pi(A) &= \sup_{u \in A} \pi(u) = \max(\pi(\text{flu}), \pi(\text{cold})) = 0.9. \\
			N(A) &= \inf_{u \notin A} (1 - \pi(u)) = 1 - \pi(\text{allergy}) = 1 - 0.3 = 0.7.
		\end{align*}
		\item Interpretation:
		\begin{itemize}
			\item Possibility (\(\Pi(A)\)): Flu or cold is highly possible (\(0.9\)).
			\item Necessity (\(N(A)\)): It is moderately certain (\(0.7\)) that the patient has either flu or cold.
		\end{itemize}
	\end{itemize}
\end{frame}

\section{Possibilistic Logic}
\begin{frame}{Basic Possibilistic Logic}
	\begin{block}{Basic possibilistic logic formula}
		A basic possibilistic logic formula is represented as:
		\[
		(a, \alpha)
		\]
		where:
		\begin{itemize}
			\item \(a\): A classical logic formula (e.g., \(p \to q\)).
			\item \(\alpha \in \left( 0, 1 \right]  \): Certainty level (minimum necessity degree of \(a\)).
		\end{itemize}
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item \((b \to f, 0.9)\): Birds fly with certainty 0.9.
			\item \((p \to \neg f, 1)\): Penguins do not fly, absolutely certain.
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Notes: 
% A basic possibilistic logic formula is a pair (a,?) made of a classical logic formula a associated with a certainty level ? ?(0,1], viewed as a lower bound of a necessity measure, i.e., (a,?) is understood as N(a) ? ?
% Interpretation
% Formula a is at least certain to degree alpha
% The higher alpha, the stronger the belief in a
% In the following, we only consider the case of (basic) possibilistic propositional logic, ?L for short, i.e., possibilistic logic formulas (a,?) are such that a is a formula in a propositional language

% Slide 9:
\begin{frame}{Axioms and Inference Rules}
	
	\begin{itemize}
		\item Certainty weakening:
		\[
		\text{If } \beta \leq \alpha, \quad (a, \alpha) \vdash (a, \beta)
		\]
		\begin{exampleblock}{Example}
		\((\text{It will rain tomorrow}, 0.9) \vdash (\text{It will rain tomorrow}, 0.7)\)  
		- If the certainty level decreases, the conclusion still holds.
		\end{exampleblock}
		\item Modus ponens:
		\[
		(\neg a \lor b, \alpha), \quad (a, \alpha) \vdash (b, \alpha)
		\]
		\begin{exampleblock}{Example}
			\((\neg \text{fever} \lor \text{flu}, 0.8), \quad (\text{fever}, 0.8) \vdash (\text{flu}, 0.8)\)  
			- If fever is observed, flu is inferred with certainty 0.8.
		\end{exampleblock}
	\end{itemize}
\end{frame}
% Notes:
% 

\begin{frame}{Axioms and Inference Rules}
	\begin{itemize}
		\item Resolution:
		\[
		(\neg a \lor b, \alpha), \quad (a \lor c, \alpha) \vdash (b \lor c, \alpha)
		\]
		\begin{exampleblock}{Example}
			\((\neg \text{fever} \lor \text{flu}, 0.8), \quad (\text{fever} \lor \text{headache}, 0.8) \vdash (\text{flu} \lor \text{headache}, 0.8)\)
		\end{exampleblock}
		\item Weakest link resolution:
		\[
		(\neg a \lor b, \alpha), \quad (a \lor c, \beta) \vdash (b \lor c, \min(\alpha, \beta))
		\]
		\begin{exampleblock}{Example}
			\((\neg \text{fever} \lor \text{flu}, 0.9), \quad (\text{fever} \lor \text{headache}, 0.7) \vdash (\text{flu} \lor \text{headache}, 0.7)\)
			- The weakest certainty (0.7) determines the conclusion.
		\end{exampleblock}
	\end{itemize}
\end{frame}

\begin{frame}{Axioms and Inference Rules}
	\begin{itemize}
		\item Formula weakening:
		\[
		\text{If } a \vdash b \text{ in classical logic, then } (a, \alpha) \vdash (b, \alpha).
		\]
		\begin{exampleblock}{Example}
			\((\text{fever}, 0.9) \vdash (\text{flu}, 0.9)\)
			- Logical implication is preserved with the same certainty level.
		\end{exampleblock}
	\end{itemize}
\end{frame}

% Slide 1: Inference in Possibilistic Logic
\begin{frame}{Inference in Possibilistic Logic}
	\begin{itemize}
		\item Knowledge Base (\(\Gamma\)):
		\[
		\Gamma = \{(a_i, \alpha_i), i = 1, \dots, m\}.
		\]
		- A set of possibilistic formulas \((a_i, \alpha_i)\), where \(a_i\) is a classical formula and \(\alpha_i\) is its certainty level.		
		\item Stratification by Certainty Levels:
		- The knowledge base \(\Gamma\) is divided into nested level cuts:
		\[
		\Gamma_\alpha = \{(a_i, \alpha_i) \in \Gamma \mid \alpha_i \geq \alpha\}.
		\]
		\item Properties:
		\begin{itemize}
			\item Nested structure: \(\Gamma_\alpha \subseteq \Gamma_\beta \quad \text{if } \beta \leq \alpha.\)
			\item Formulas in \(\Gamma_\alpha\) contribute to inferences at certainty level \(\alpha\).
		\end{itemize}
	\end{itemize}
\end{frame}

% Slide 3: Inconsistency Level
\begin{frame}{Inconsistency Level of a Knowledge Base}
	\begin{itemize}
		\item Definition:
		- The inconsistency level of \(\Gamma\) is:
		\[
		\text{inc-l}(\Gamma) = \max\{\alpha \mid \Gamma \vdash (\bot, \alpha)\}.
		\]
		- It is the highest certainty level at which the knowledge base becomes inconsistent.
		\vspace{0.3cm}
		
		\item Safe Formulas:
		- Formulas with certainty levels \(> \text{inc-l}(\Gamma)\) are safe from inconsistency.
		\[
		\text{inc-l}(\{ (a_i, \alpha_i) \mid (a_i, \alpha_i) \in \Gamma \text{and } \alpha_i > \text{inc-l}(\Gamma)\}) = 0.
		\]
		
		\item Key Property:
		- If \((\Gamma)^*\) (the classical counterpart of \(\Gamma\)) is consistent, then:
		\[
		\text{inc-l}(\Gamma) = 0.
		\]
	\end{itemize}
\end{frame}



% Slide 5: Handling Inconsistencies
\begin{frame}{Managing Inconsistent Information}
	\begin{itemize}
		\item Possibilistic logic can handle conflicting information by:
		\begin{itemize}
			\item Identifying an **inconsistency level** (\(inc-l\)).
			\item Filtering out rules with certainty levels below \(inc-l\).
		\end{itemize}
		\item Example:
		\[
		\{(a, 0.8), (\neg a, 0.7)\}
		\]
		\begin{itemize}
			\item Inconsistency level: \(inc-l = 0.7\).
			\item Filtered result: \((a, 0.8)\) dominates.
		\end{itemize}
	\end{itemize}
\end{frame}

\section{Applications}
\begin{frame}{Default reasoning}
	\begin{itemize}
		\item Default rule:
		\[
		\Pi (a \land b) > \Pi(a \land \neg b)
		\]
		- \(a \to b\): "If \(a\), then \(b\) is more plausible than \(\neg b\)."
		\item In terms of necessity:
		\[
		N(b | a) = 1 - \Pi(\neg b | a) > 0
		\]
		\item Comparative relation:
		\[
		a \land b >_\Pi a \land \neg b
		\]
	\end{itemize}
\end{frame}
% It expresses that in the context where ais true, there exists situations where having b true is strictly more satisfactory than any situations where b is false in the same context.


% Slide 3: Penguin Example - Rules and Constraints
\begin{frame}{Penguin Example: Rules and Constraints}
	\begin{itemize}
		\item Default Rules:
		\begin{align*}
			d_1 &: b \rightarrow f \quad (\text{"Birds fly"}), \\
			d_2 &: p \rightarrow \neg f \quad (\text{"Penguins do not fly"}), \\
			d_3 &: p \rightarrow b \quad (\text{"Penguins are birds"}).
		\end{align*}
		\item Constraints:
		\begin{align*}
			C_1 &: b \land f >_\Pi b \land \neg f \quad \text{(birds fly)}. \\
			C_2 &: p \land \neg f >_\Pi p \land f \quad \text{(penguins don't fly)}. \\
			C_3 &: p \land b >_\Pi p \land \neg b \quad \text{(penguins are birds)}.
		\end{align*}
		\item Principle of Minimal Specificity: Assigns possibility levels to worlds while satisfying these constraints.
	\end{itemize}
\end{frame}

% Slide 4: Penguin Example - Worlds and Possibility
\begin{frame}{Penguin Example: Worlds and Possibility}
	\begin{itemize}
		\item Worlds (\(\Omega\)):
		\[
		\Omega = \{\omega_0, \omega_1, \omega_2, \omega_3, \omega_4, \omega_5, \omega_6, \omega_7\},
		\]
		where each \(\omega_i\) represents a unique combination of \(b, f, p\) (bird, fly, penguin), e.g: $  \omega_0 : \neg b \land \neg f \land \neg p, \quad
		\omega_1 : \neg b \land \neg f \land p, \quad
		\omega_2 : \neg b \land f \land \neg p, \quad
		\omega_3 : \neg b \land f \land p, \quad
		\omega_4 : b \land \neg f \land \neg p, \quad
		\omega_5 : b \land \neg f \land p, \quad
		\omega_6 : b \land f \land \neg p, \quad
		\omega_7 : b \land f \land p $
		\vspace{0.3cm}
		\item Constraints:
		\begin{align*}
			C_1 &: \max(\pi(\omega_6), \pi(\omega_7)) > \max(\pi(\omega_4), \pi(\omega_5)), \\
			C_2 &: \max(\pi(\omega_5), \pi(\omega_1)) > \max(\pi(\omega_3), \pi(\omega_7)), \\
			C_3 &: \max(\pi(\omega_5), \pi(\omega_7)) > \max(\pi(\omega_1), \pi(\omega_3)).
		\end{align*}
	\end{itemize}
\end{frame}
% The constraints C_1, C_2, C_3 determine relationships between the possibility degrees of the worlds (\pi(\omega_i)). These constraints ensure that some worlds are ranked higher than others.
% The partition respects these rules by ranking worlds consistently with their plausibility under the defaults

\begin{frame}{Constraints and Plausibility Relationships}
	\begin{itemize}
		\item Default Rule \(d_1: b \to f\) ("Birds fly"):
		\begin{itemize}
			\item Makes worlds where birds fly (\(b \land f\)) more plausible than those where birds don?t fly (\(b \land \neg f\)).
			\item Affects:
			\[
			\omega_6 \, (\text{bird flies}) \quad \text{vs.} \quad \omega_4, \omega_5 \, (\text{bird doesn?t fly}).
			\]
		\end{itemize}
		\vspace{0.2cm}
		
		\item Default Rule \(d_2: p \to \neg f\) ("Penguins don?t fly"):
		\begin{itemize}
			\item Makes worlds where penguins don?t fly (\(p \land \neg f\)) more plausible than those where penguins fly (\(p \land f\)).
			\item Affects:
			\[
			\omega_5 \, (\text{penguin doesn?t fly}) \quad \text{vs.} \quad \omega_7, \omega_3 \, (\text{penguin flies}).
			\]
		\end{itemize}
		\vspace{0.2cm}
		
		\item*Default Rule \(d_3: p \to b\) ("Penguins are birds"):
		\begin{itemize}
			\item Ensures penguins are considered birds, influencing how penguin-related worlds (\(p\)) are ranked.
		\end{itemize}
	\end{itemize}
\end{frame}

\begin{frame}{Penguin Example: Worlds and Possibility}
	\begin{itemize}
		\item Partition: Worlds are ranked based on constraints:
		\[
		\{\omega_0, \omega_2, \omega_6\} >_\Pi \{\omega_4, \omega_5\} >_\Pi \{\omega_1, \omega_3, \omega_7\}.
		\]
		\vspace{0.3cm}
		\item \(m\) is the number of distinct plausibility levels (possibility degrees) assigned to the worlds in \(\Omega\).
		\item Possibility Levels:
		\[
		\pi(\omega_0) = \pi(\omega_2) = \pi(\omega_6) = 1, \quad \pi(\omega_4) = \pi(\omega_5) = 2/3, \quad \pi(\omega_1) = \pi(\omega_3) = \pi(\omega_7) = 1/3.
		\]
	\end{itemize}
\end{frame}
% Worlds in this group are most plausible under the constraints.
% From the possibility distribution ? associated with the well ordered partition, we can compute the necessity level N(a) of any proposition a. The method then consists in turning each default pi qi into a possibilistic clause (¬pi?qi, N(¬pi?qi)), where N is computed from the greatest possibility distribution ? induced by the set of constraints corresponding to the default knowledge base, as already explained. We thus obtain a possibilistic logic base K
\begin{frame}{Computing Necessity Measures and Possibilistic Base}
	\begin{itemize}
		\item Necessity Measure (\(N(A)\)):
		\[
		N(A) = \inf_{u \notin A} (1 - \pi(u)).
		\]
		
		\item Computation:
		\begin{itemize}
			\item \(N(\neg p \lor \neg f)\):
			\[
			N(\neg p \lor \neg f) = \min\{1 - \pi(\omega_3), 1 - \pi(\omega_7)\} = \min\{1 - 1/3, 1 - 1/3\} = 2/3.
			\]
			\item \(N(\neg b \lor f)\):
			\[
			N(\neg b \lor f) = \min\{1 - \pi(\omega_4), 1 - \pi(\omega_5)\} = \min\{1 - 2/3, 1 - 2/3\} = 1/3.
			\]
			\item \(N(\neg p \lor b)\):
			\[
			N(\neg p \lor b) = \min\{1 - \pi(\omega_1), 1 - \pi(\omega_3)\} = \min\{1 - 1/3, 1 - 1/3\} = 2/3.
			\]
		\end{itemize}
		
		\item Resulting Possibilistic Logic Base (\(K\)):
		\[
		K = \{(\neg p \lor \neg f, 2/3), (\neg p \lor b, 2/3), (\neg b \lor f, 1/3)\}.
		\]
	\end{itemize}
\end{frame}

\begin{frame}{Reasoning with Tweety: Bird and Penguin Cases}
	\begin{itemize}
		\item Initial Information (Tweety is a Bird):
		\[
		F = \{(b, 1)\} \quad \text{(Tweety is a bird with certainty 1)}.
		\]
		- Apply the weakest link resolution rule:
		\[
		K \cup \{(b, 1)\} \vdash (f, 1/3).
		\]
		- Conclusion: If Tweety is a bird, we conclude that Tweety flies with certainty \( \frac{1}{3} \).
		
		\item New Information (Tweety is a Penguin):
		\[
		F = \{(b, 1), (p, 1)\} \quad \text{(Tweety is a bird and a penguin with certainty 1)}.
		\]
		- Apply the weakest link resolution rule:
		\[
		K \cup F \vdash (\bot, 1/3).
		\]
		- Conclusion: The knowledge base becomes inconsistent with certainty \( \frac{1}{3} \).
		
		\item Final Conclusion (Tweety is a Penguin):
		- Despite inconsistency, we can still infer:
		\[
		K \cup F \vdash (\neg f, 2/3).
		\]
		- Conclusion: Since Tweety is a penguin, we conclude that Tweety does not fly with certainty \( \frac{2}{3} \).
	\end{itemize}
\end{frame}
% This encoding takes advantage of the fact that when a new piece of information is received, the level of inconsistency of the base cannot decrease, and if it strictly increases, some inferences that were safe before are now drowned in the new inconsistency level of the base and are thus no longer allowed, hence a nonmonotonic consequence mechanism takes place.
