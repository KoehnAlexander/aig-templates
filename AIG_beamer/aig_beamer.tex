\section{Possibility Theory}
% Now I will introduce you first into Possibilty theory

\begin{frame}{Possibility Distribution}
	\begin{block}{Possibility distribution}
		A \textbf{possibility distribution} is a mapping:
		\[
		\pi: U \to S,
		\]
		where:
		\begin{itemize}
			\item \( U \): Set of states
			\item \( S \): Totally ordered scale (\( S = [0, 1] \) or finite chains).
			\item \( \pi(u) \): Degree of possibility (plausibility) of state \( u \).
		\end{itemize}
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item \( \pi(u) = 0 \): State \( u \) is \textbf{impossible}.
			\item \( \pi(u) = 1 \): State \( u \) is \textbf{fully possible}.
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Slide 5: Possibility Distribution
% Notes:
% To start with possibility theory we need to first understand possibility distributions
% A possibility distribution is a mapping pi from U to a totally ordered scale S, with top denoted by 1 and bottom by 0
% For the further explanations in this presentation we will use the ordered scale 0 to 1
% The function pi represents the state of knowledge of an agents
% it distinguishes what is plausible from what is less plausible

\begin{frame}{Possibility and Necessity Functions}
	\begin{block}{Possibility measure}
		\[
		\Pi(A) = \sup_{u \in A} \pi(u).
		\]
		Evaluates the extent to which \( A \) is consistent with \( \pi \).
	\end{block}
	\begin{block}{Necessity measure}
		\[
		N(A) = \inf_{u \notin A} (1 - \pi(u)).
		\]
		Evaluates the extent to which \( A \) is certainly implied by \( \pi \).
	\end{block}
	
\end{frame}
% Slide 6: Possibility and Necessity Functions
% Notes:
% If we have the question:
% Does event A occur? 
% A is a subset of states
% We can respond in possibility theory with two measures calculated from the possibility distribution
% The Possibility evaluates to which extent A is consistent with pi.
% It is the maximun of possibility of all the states of A
% The Necessity evaluates to which extent A is certainly implied by pi.
% It is the minimum of the inversed possibility value of all states which are not in A

\begin{frame}{Example}
	\begin{itemize}
		\item A doctor is trying to diagnose a patient based on symptoms and diseases:
		\begin{itemize}
			\item \( U = \{\text{flu}, \text{cold}, \text{allergy}\} \).
			\item Possibility distribution \(\pi\) based on symptom severity:
			\[
			\pi(\text{flu}) = 0.9, \quad \pi(\text{cold}) = 0.6, \quad \pi(\text{allergy}) = 0.3.
			\]
		\end{itemize}
		\item The doctor evaluates the possibility and necessity measures for a hypothesis \( A \):
		\[ 
		A = \{\text{flu}, \text{cold}\}
		\]
		\item Calculation:
		\begin{align*}
			\Pi(A) &= \sup_{u \in A} \pi(u) = \max(\pi(\text{flu}), \pi(\text{cold})) = 0.9. \\
			N(A) &= \inf_{u \notin A} (1 - \pi(u)) = 1 - \pi(\text{allergy}) = 1 - 0.3 = 0.7.
		\end{align*}
		\item Interpretation:
		\begin{itemize}
			\item Possibility (\(\Pi(A)\)): Flu or cold is highly possible (\(0.9\)).
			\item Necessity (\(N(A)\)): It is moderately certain (\(0.7\)) that the patient has either flu or cold.
		\end{itemize}
	\end{itemize}
\end{frame}

\section{Possibilistic Logic}

\begin{frame}{Basic Possibilistic Logic}
	\begin{block}{Basic possibilistic logic formula}
		A basic possibilistic logic formula is represented as:
		\[
		(a, \alpha)
		\]
		where:
		\begin{itemize}
			\item \(a\): A classical logic formula (e.g., \(p \to q\)).
			\item \(\alpha \in \left( 0, 1 \right]  \): Certainty level (minimum necessity degree of \(a\)).
		\end{itemize}
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item \((b \to f, 0.9)\): Birds fly with certainty 0.9.
			\item \((p \to \neg f, 1)\): Penguins do not fly, absolutely certain.
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Slide 9: Basic Possibilistic Logic
% Notes: 
% After we have now seen the core concept of possibility theory we will jump in the next chapter right into Possibilistic Logic.
% A basic possibilistic logic formula is a pair a and alpha:
% a is a classical logic formula
% and alpha is the  associated certainty level. 
% it can be viewed as a lower bound of a necessity measure,
% In example (a, alpha) is understood as N(a) > alpha
% We can interpret that the Formula a is at least certain to degree alpha
% The higher alpha, the stronger the belief in a
% For Example we can model that Birds fly with Certainty 0.9

\begin{frame}{Axioms and Inference Rules: Certainty Weakening}
	\begin{block}{Certainty weakening}
		\[
		\text{If } \beta \leq \alpha, \quad (a, \alpha) \vdash (a, \beta)
		\]
	\end{block}
	\begin{exampleblock}{Example}
		\((\text{It will rain tomorrow}, 0.9) \vdash (\text{It will rain tomorrow}, 0.7)\)  \\
		If the certainty level decreases, the conclusion still holds.
	\end{exampleblock}
	
\end{frame}
% Slide 10: Axioms and Inference Rules: Certainty Weakening
% Notes:
% The axioms of Possibilistic Logic are those of propositional logic
% In the following I will explain you the inference rules of possibilistic logic, which we will need later in our example
% The Certainty weakening rule states that:
% If a formula is valid with a high certainty level, it is also valid with any lower certainty level
% Certainty levels represent the minimum degree of belief or plausibility with which a formula holds.
% If a formula holds with a certainty level alpha, it automatically implies that it holds with any lower level of certainty beta because we are being less demanding.

% -| indicates that a statement (or formula) can be derived or proven from a set of premises in a formal system.
% arrow: Implies (if-then).

\begin{frame}{Axioms and Inference Rules: Modus Ponens}
	\begin{block}{Modus ponens}
		\[
		(\neg a \lor b, \alpha), (a, \alpha) \vdash (b, \alpha), \quad \forall \alpha \in (0, 1]
		\]
	\end{block}
	\begin{exampleblock}{Example}
		\((\neg \text{fever} \lor \text{flu}, 0.8), (\text{fever}, 0.8) \vdash (\text{flu}, 0.8)\)  \\
		If fever is observed, flu is inferred with certainty 0.8.
	\end{exampleblock}
\end{frame}
% Slide 11: Axioms and Inference Rules: Modus Ponens
% Notes:
% The Modus ponens rules states that:
% If a to b is true with certainty level alpha, and a is also true with certainty level alpha
% Then b must be true with certainty level alpha.

\begin{frame}{Axioms and Inference Rules: Resolution}
	\begin{block}{Resolution}
		\[
		(\neg a \lor b, \alpha), \quad (a \lor c, \alpha) \vdash (b \lor c, \alpha)
		\]
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item Propositions:
			\begin{itemize}
				\item \(a = \text{"It is raining"}\),
				\item \(b = \text{"The ground is wet"}\),
				\item \(c = \text{"There is a thunderstorm"}\).
			\end{itemize}
			\item Resolution:
			\[
			(\neg a \lor b, 0.7), (a \lor c, 0.7) \vdash ( b \lor c, 0.7)
			\]
			\item Interpretation: "The ground is wet or there is a thunderstorm" holds with certainty \(0.7\).
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Slide 12: Axioms and Inference Rules: Resolution
% Notes:
% In classical logic, the resolution rule allows us to combine two clauses that share a literal to eliminate that literal and infer a new clause.
% In possibilistic logic, the resolution rule preserves the certainty level alpha, because the inferred formula (b or c) cannot be more certain than the least certain premise

\begin{frame}{Axioms and Inference Rules: Weakest Link Resolution}
	\begin{block}{Weakest link resolution}
		\[
		(\neg a \lor b, \alpha), \quad (a \lor c, \beta) \vdash (b \lor c, \min(\alpha, \beta))
		\]
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item Premises:
			\[
			(\neg a \lor b, 0.7), \quad (a \lor c, 0.5).
			\]
			\item Inference:
			\[
			(\neg a \lor b, 0.7), \quad (a \lor c, 0.5) \vdash (b \lor c, \min(0.7, 0.5)).
			\]
			\item Interpretation: "The ground is wet or there is a thunderstorm" is true with certainty \(0.5\), limited by the weaker premise.
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Slide 13: Axioms and Inference Rules: Weakest Link Resolution
% Notes: 
% The weakest link resolution rule extends the resolution rule by  the concept of different certainty levels alpha and beta in the premises.
% The weakest link resolution reflects the idea that the certainty of the conclusion cannot exceed the weakest certainty of the premises.
% If one of the premises is less certain, it becomes the bottleneck for the certainty of the conclusion

\begin{frame}{Axioms and Inference Rules: Formula Weakening}
	\begin{block}{Formula weakening}
		\[
		\text{If } a \vdash b \text{ in classical logic, then } (a, \alpha) \vdash (b, \alpha).
		\]
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item Premise:
			\[
			(a, 0.8) \quad \text{("It is raining with certainty 0.8")}.
			\]
			\item Conclusion:
			\[
			(b, 0.8) \quad \text{("The ground is wet with certainty 0.8")}.
			\]
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Slide 14: Axioms and Inference Rules: Formula Weakening
% Notes:
% The Formula weakening rule states that if b can be derived from a in classical logic
% It can be derived in possibilistic logic with the same certainty level

\begin{frame}{Knowledge Base}
	\begin{block}{Knowledge base (\(\Gamma\))}
		\[
		\Gamma = \{(a_i, \alpha_i), i = 1, \dots, m\}
		\]
	\end{block}
	\begin{block}{Knowledge base (\( \Gamma_\alpha \))}
		The knowledge base \(\Gamma\) is divided into nested level cuts:
		\[
		\Gamma_\alpha = \{(a_i, \alpha_i) \in \Gamma \mid \alpha_i \geq \alpha\}
		\]
		\begin{itemize}
		\item Nested structure: \(\Gamma_\alpha \subseteq \Gamma_\beta \quad \text{if } \beta \leq \alpha.\)
		\end{itemize}
	\end{block}
\end{frame}
% Slide 15: Knowledge Base
% Notes:
% We can define a possibilistic knowledge base as a collection of possibilistic formulas
% Further we can define the alpha cut of knowledge base
% It only contains those formulas from the origin knowledge base with a certainty level of at least alpha
% It has a nested structure where the alpha knwoledge base is a part of the beta knowledge base if alpha is higher than beta

\begin{frame}{Inference in Possibilistic Logic}
	\begin{block}{Inference}
		\[
		\Gamma \vdash (a, \alpha) \iff \Gamma_\alpha \vdash (a, \alpha)
		\]
		\begin{itemize}
		\item A possibilistic inference \((a, \alpha)\) depends only on formulas with certainty levels \(\geq \alpha\).
		\end{itemize}
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item Knowledge Base:
			\[
			\Gamma = \{(a_1, 0.9), (a_2, 0.7), (a_3, 0.5)\}.
			\]
			\item Inference for \(\alpha = 0.7\):
			\[
			\Gamma_\alpha \vdash (a_2, 0.7) \implies \Gamma \vdash (a_2, 0.7).
			\]
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Slide 16: Inference in Possibilistic Logic
% Notes:
% If we want to infer from the knowledge, we can only look on the formulas with a certainty of at least alpha
% This means that a inference is ony valid if it is valid with the alpha cut of a knowledge base

\begin{frame}{Inconsistency Level of a Knowledge Base}
	\begin{block}{Inconsistency level of \(\Gamma\)}
		\[
		\text{inc-l}(\Gamma) = \max\{\alpha \mid \Gamma \vdash (\bot, \alpha)\}.
		\]
	\end{block}
	\begin{exampleblock}{Properties}
		Formulas with certainty levels \(> \text{inc-l}(\Gamma)\) are safe from inconsistency.
		\[
		\text{inc-l}(\{ (a_i, \alpha_i) \mid (a_i, \alpha_i) \in \Gamma \text{and } \alpha_i > \text{inc-l}(\Gamma)\}) = 0
		\]
	\end{exampleblock}
\end{frame}
% Slide 17: Inconsistency Level of a Knowledge Base
% Notes:
% An Important attribute of knowledge bases is the inconsistency level
% It is the highest certainty level at which the knowledge base becomes inconsistent
% the properties of the inconsistency level shows that all formulas which have a higher certainty level than the inconsistency level are safe form inconsistency

\section{Applications}

\begin{frame}{Default Reasoning}
	\begin{block}{Default rule}
		\[
		\Pi (a \land b) > \Pi(a \land \neg b)
		\]
		"If \(a\), then \(b\) is more plausible than \(\neg b\)."
	\end{block}
	\begin{block}{Comparative possibility relation}
		\[
		a \land b >_\Pi a \land \neg b
		\]
	\end{block}
\end{frame}
% Slide 19: Default Reasoning
% Notes:
% One application of possibilistic logic is Default Reasoning: 
% where we reason with incomplete information.
% In default reasoning we can understand default rule "if a then b, generally" 
% that if a, then b is more plausible than not b.
% This is expressed in the operator
% the above constraint can be equivantly expressed in terms of comparative possibility relation

\begin{frame}{Penguin Example: Rules and Constraints}
	\begin{exampleblock}{Default Rules}
		\begin{align*}
			d_1 &: b \rightarrow f \quad (\text{"Birds fly"}), \\
			d_2 &: p \rightarrow \neg f \quad (\text{"Penguins do not fly"}), \\
			d_3 &: p \rightarrow b \quad (\text{"Penguins are birds"}).
		\end{align*}
	\end{exampleblock}
	\begin{itemize}
		\item Constraints:
		\begin{align*}
			C_1 &: b \land f >_\Pi b \land \neg f \quad \text{(birds fly)}. \\
			C_2 &: p \land \neg f >_\Pi p \land f \quad \text{(penguins don't fly)}. \\
			C_3 &: p \land b >_\Pi p \land \neg b \quad \text{(penguins are birds)}.
		\end{align*}
	\end{itemize}
\end{frame}
% Slide 20: Penguin Example: Rules and Constraints
% Notes:
% With the gained knowledge we can jump into the Penguin example
% It has three default rules
% Birds fly
% Penguins do not fly
% Penguins are birds

\begin{frame}{Penguin Example: Worlds and Possibility}
	\begin{block}{Worlds (\(\Omega\))}
		\(\Omega\) is the finite set of interpretations of the considered propositional language.
	\end{block}
	\begin{exampleblock}{Worlds (\(\Omega\))}
		Worlds (\(\Omega\)):
		\[
		\Omega = \{\omega_0, \omega_1, \omega_2, \omega_3, \omega_4, \omega_5, \omega_6, \omega_7\},
		\]
		where each \(\omega_i\) represents a unique combination of \(b, f, p\) (bird, fly, penguin):
		\begin{align*}
			\omega_0 &: \neg b \land \neg f \land \neg p, \quad  \omega_1 : \neg b \land \neg f \land p,  \\
			\omega_2 &: \neg b \land f \land \neg p, \quad 	\omega_3 : \neg b \land f \land p,  \\
			\omega_4 &: b \land \neg f \land \neg p, \quad  \omega_5 : b \land \neg f \land p,  \\
			\omega_6 &: b \land f \land \neg p, \quad  \omega_7 : b \land f \land p 
		\end{align*} 
	\end{exampleblock}
\end{frame}
% Slide 21: Penguin Example: Worlds and Possibility
% Notes:
% Now we can define the worlds omega, which correspond to the interpretation of the corresponding language
% In this case birds, penguins and fly
% each omega i represents a unique combination

\begin{frame}{Penguin Example: Updated Constraints}
	\begin{itemize}
		\item Constraints:
		\begin{align*}
			C_1 &: \max(\pi(\omega_6), \pi(\omega_7)) > \max(\pi(\omega_4), \pi(\omega_5)), \\
			C_2 &: \max(\pi(\omega_5), \pi(\omega_1)) > \max(\pi(\omega_3), \pi(\omega_7)), \\
			C_3 &: \max(\pi(\omega_5), \pi(\omega_7)) > \max(\pi(\omega_1), \pi(\omega_3)).
		\end{align*}
	\end{itemize}
\end{frame}
% Slide 22: Penguin Example: Updated Constraints
% Notes:
% we can now update the constraints with including the possible worlds
% These constraints ensure that some worlds are ranked higher than others.

\begin{frame}{Penguin Example: Partition and Possibility Levels}
	\begin{exampleblock}{Partition}
		Worlds are ranked based on constraints:
		\[
		\{\omega_0, \omega_2, \omega_6\} >_\Pi \{\omega_4, \omega_5\} >_\Pi \{\omega_1, \omega_3, \omega_7\}.
		\]
	\end{exampleblock}
	\begin{exampleblock}{Possibility Levels}
		\begin{align*}
			\pi(\omega_0) &= \pi(\omega_2) = \pi(\omega_6) = 1, \\
			\pi(\omega_4) &= \pi(\omega_5) = 2/3, \\
			\pi(\omega_1) &= \pi(\omega_3) = \pi(\omega_7) = 1/3
		\end{align*}
	\end{exampleblock}
\end{frame}
% Slide 23: Penguin Example: Partition and Possibility Levels
% Notes
% By using the principle of minimal specificity, we can assign possibility levels to the worlds with satisfying the constraints
% Worlds in highest group are most plausible under the constraints.
% We have three different possibility levels
% We can now assign the worlds possibilities with using the number of possibility levels

\begin{frame}{Penguin Example: Computing Necessity Measures}
	\begin{block}{Necessity Measure (\(N(A)\))}
		\[
		N(A) = \inf_{u \notin A} (1 - \pi(u)).
		\]
	\end{block}
	\begin{exampleblock}{Computation}
		\begin{align*}
			N(\neg p \lor \neg f) &= \min\{1 - \pi(\omega_3), 1 - \pi(\omega_7)\} = \min\{1 - 1/3, 1 - 1/3\} = 2/3 \\
			N(\neg b \lor f) &= \min\{1 - \pi(\omega_4), 1 - \pi(\omega_5)\} = \min\{1 - 2/3, 1 - 2/3\} = 1/3
			\\
			N(\neg p \lor b) &= \min\{1 - \pi(\omega_1), 1 - \pi(\omega_3)\} = \min\{1 - 1/3, 1 - 1/3\} = 2/3
		\end{align*}
	\end{exampleblock}
\end{frame}
% Slide 24: Penguin Example: Computing Necessity Measures
% Notes:
% From the possibility of the worlds we can now compute the necessity level N of any proposition
% And have as a result for each rule a certainty level


\begin{frame}{Penguin Example: Resulting Possibilistic Logic Base}
	\begin{exampleblock}{Resulting Possibilistic Logic Base (\(K\))}
		\[
		K = \{(\neg p \lor \neg f, 2/3), (\neg p \lor b, 2/3), (\neg b \lor f, 1/3)\}.
		\]
	\end{exampleblock}
\end{frame}
% Slide 25: Penguin Example: Resulting Possibilistic Logic Base
% Notes:
% We can now build a possibilistic logic base K
% K encodes the generic knowledge embedded in the default rules.

\begin{frame}{Penguin Example: Reasoning with Initial Information}
	Initial Information (Tweety is a Bird):
	\[
	F = \{(b, 1)\} \quad \text{(Tweety is a bird with certainty 1)}.
	\]
	\begin{itemize}
		\item From the rule \((\neg b \lor f, 1/3)\) in \(K\), we know:
		\[
		(\neg b \lor f, 1/3) \quad \text{and} \quad (b, 1) \vdash (f, \min(1, 1/3)) = (f, 1/3).
		\]
		\item $	K \cup \{(b, 1)\} \vdash (f, 1/3) $
		\item  Conclusion: If Tweety is a bird, we conclude that Tweety flies with certainty \( \frac{1}{3} \).
	\end{itemize}
\end{frame}
% Slide 26: Penguin Example: Reasoning with Initial Information
% Notes:
% With the initial information that tweety is fully certain a bird
% We can infer from the logic base by applying the weakest link resolution rule and modus ponens
% We can conclude that Tweety flies with certainty one third

\begin{frame}{Penguin Example: Reasoning with New Information}
	New Information (Tweety is a Penguin):
	\[
	F = \{(b, 1), (p, 1)\} \quad \text{(Tweety is a bird and a penguin with certainty 1)}.
	\]
	\begin{itemize}
		\item From the rule \((\neg p \lor \neg f, 2/3)\) in \(K\), we know:
		\[
		(\neg p \lor \neg f, 2/3) \quad \text{and} \quad (p, 1) \vdash (\neg f, \min(1, 2/3)) = (\neg f, 2/3).
		\]
		\item From the rule \((\neg b \lor f, 1/3)\) in \(K\) and the new information \((p, 1)\), we know :
		\[
		(\neg b \lor f, 1/3) \quad \text{and} \quad (b, 1) \vdash (f, \min(1, 1/3))= (f, 1/3).
		\]
		\item These results conflict because:
		\[
		(\neg f, 2/3) \quad \text{and} \quad (f, 1/3) \vdash (\bot, \min(2/3, 1/3)) = (\bot, 1/3).
		\]
		\item $ K \cup F \vdash (\bot, 1/3) $
		\item Conclusion: The knowledge base becomes inconsistent with certainty \( \frac{1}{3} \).
	\end{itemize}
\end{frame}
% Slide 27: Penguin Example: Reasoning with New Information
% Notes:
% With new information that Tweety is a penguin
% We apply both informaiton to the knowledge base
% Now we get a conflict
% And thus our knowledge base becomes inconsistent


\begin{frame}{Penguin Example: Reasoning Final Conclusion}	
	Final Conclusion (Tweety is a Penguin):
	\begin{itemize}				
		\item Despite inconsistency, we can still infer:
		\[
		K \cup F \vdash (\neg f, 2/3).
		\]
		\[
		(\neg p \lor \neg f, 2/3) \quad \text{and} \quad (p, 1) \vdash (\neg f, \min(1, 2/3)) = (\neg f, 2/3).
		\]
		\item Conclusion: Since Tweety is a penguin, we conclude that Tweety does not fly with certainty \( \frac{2}{3} \).
	\end{itemize}
\end{frame}

\begin{frame}{Specificity of a Possibility Distribution}
	\begin{block}{Specificity of a Possibility Distribution}
		A distribution \( \pi \) is \textbf{at least as specific} as \( \pi' \) if:
		\[
		\pi(u) \leq \pi'(u), \quad \forall u \in U.
		\]
	\end{block}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item \textbf{Complete knowledge:} \( \pi(u_0) = 1 \), \( \pi(u) = 0 \) for all \( u \neq u_0 \).
			\item \textbf{Complete ignorance:} \( \pi(u) = 1 \) for all \( u \in U \).
		\end{itemize}
	\end{exampleblock}
	\begin{block}{Principle of minimal specificity}
		It states that any hypothesis not known to be impossible cannot be ruled out. It is a minimal commitment, cautious, information principle.
	\end{block}
\end{frame}

% Slide 5: Key Properties of Measures
\begin{frame}{Key Properties of Measures}
	\begin{block}{Duality Relationship}
		\[
		N(A) = 1 - \Pi(A^c).
		\]
		Where \( A^c \) is the complement of \( A \).
	\end{block}
	\begin{block}{Possibility: Maxitivity Property}
		\[
		\Pi(A \cup B) = \max(\Pi(A), \Pi(B)).
		\]
	\end{block}
	\begin{block}{Necessity: Minimization Property}
		\[
		N(A \cap B) = \min(N(A), N(B)).
		\]
	\end{block}
\end{frame}
% Notes:
% Possibility measures satisfy the characteristic ?maxitivity? property
% Necessity measures satisfy an axiom dual to that of possibility measures
% N(a?b) = min(N(a),N(b)), expresses that to be certain about a?b, one has to be certain about a and to be certain about b.
% Note also that we only have N(A?B) ? max(N(A),N(B)).This goes well with the idea that one may be certain about the event A?B, without being really certain about more specific events such as A and B.

% Slide 6: Certainty Qualification in Possibilistic Logi
\begin{frame}{Certainty Qualification in Possibilistic Logic}
	\begin{itemize}
		\item Certainty-qualified information: "A is certain to degree \(\alpha\)" is modeled by the constraint: 
		\[
		N(A) \geq \alpha
		\]
		\item Least Specific possibility distribution:
		\[
		\pi_{(A, \alpha)}(u) = 
		\begin{cases} 
			1 & \text{if } u \in A, \\ 
			1 - \alpha & \text{otherwise}.
		\end{cases}
		\]
	\end{itemize}
	\begin{exampleblock}{Example}
		\begin{itemize}
			\item If \(\alpha = 1\): \(\pi(A, \alpha)\) becomes the characteristic function of \(A\).
			\item If \(\alpha = 0\): \(\pi(A, \alpha)\) represents total ignorance.
		\end{itemize}
	\end{exampleblock}
\end{frame}
% Notes:
% Human knowledge is often expressed as statements with degrees of belief.
% Represents a family of possible epistemic states \(\pi\) that satisfy this constraint.
% Integral to the semantics of possibilistic logic.

% Slides 23: Penguin Example: Constraints and Plausibility Relationships
\begin{frame}{Penguin Example: Constraints and Plausibility Relationships}
	\begin{itemize}
		\item Default Rule \(d_1: b \to f\) ("Birds fly"):
		\begin{itemize}
			\item Makes worlds where birds fly (\(b \land f\)) more plausible than those where birds do not fly (\(b \land \neg f\)).
			\item Affects:
			\[
			\omega_6 \, (\text{bird flies}) \quad \text{vs.} \quad \omega_4, \omega_5 \, (\text{bird does not fly}).
			\]
		\end{itemize}
		\vspace{0.2cm}
		
		\item Default Rule \(d_2: p \to \neg f\) ("Penguins do not fly"):
		\begin{itemize}
			\item Makes worlds where penguins do not fly (\(p \land \neg f\)) more plausible than those where penguins fly (\(p \land f\)).
			\item Affects:
			\[
			\omega_5 \, (\text{penguin does not fly}) \quad \text{vs.} \quad \omega_7, \omega_3 \, (\text{penguin flies}).
			\]
		\end{itemize}
		\vspace{0.2cm}
		
		\item Default Rule \(d_3: p \to b\) ("Penguins are birds"):
		\begin{itemize}
			\item Ensures penguins are considered birds, influencing how penguin-related worlds (\(p\)) are ranked.
		\end{itemize}
	\end{itemize}
\end{frame}
